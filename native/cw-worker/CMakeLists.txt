cmake_minimum_required(VERSION 3.20)
project(xlog_cw_worker LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_BUILD_RPATH "@loader_path/../Frameworks")
set(CMAKE_INSTALL_RPATH "@loader_path/../Frameworks")
find_path(ONNXRUNTIME_INCLUDE_DIR onnxruntime_cxx_api.h PATH_SUFFIXES onnxruntime onnxruntime/core/session REQUIRED)
find_library(ONNXRUNTIME_LIBRARY onnxruntime REQUIRED)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set(MODEL_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../vendor/deepcw-engine")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate_contract.py"
  "${MODEL_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/model_contract.h" COMMAND_ERROR_IS_FATAL ANY)
find_package(Threads REQUIRED)
add_library(cw_engine engine.cpp dsp.cpp skimmer.cpp)
target_include_directories(cw_engine PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}" "${ONNXRUNTIME_INCLUDE_DIR}")
target_link_libraries(cw_engine PUBLIC "${ONNXRUNTIME_LIBRARY}" Threads::Threads)
if(APPLE)
  target_link_libraries(cw_engine PUBLIC "-framework Accelerate")
endif()
target_compile_options(cw_engine PRIVATE -Wall -Wextra -Werror)
add_executable(xlog-cw-worker main.cpp)
target_link_libraries(xlog-cw-worker PRIVATE cw_engine)
add_executable(cw-engine-tests tests.cpp)
target_link_libraries(cw-engine-tests PRIVATE cw_engine)
add_executable(cw-load-profile load_profile.cpp)
target_link_libraries(cw-load-profile PRIVATE cw_engine)
enable_testing()
add_test(NAME cw-engine COMMAND cw-engine-tests "${MODEL_DIR}/model.onnx")
if(APPLE)
  add_library(cw_process STATIC ../../Sources/CCWProcess/CCWProcess.c)
  target_include_directories(cw_process PUBLIC ../../Sources/CCWProcess/include)
  target_compile_options(cw_process PRIVATE -Wall -Wextra -Werror)
  add_executable(cw-process-probe process_probe.cpp)
  add_executable(cw-process-tests process_tests.cpp)
  target_link_libraries(cw-process-tests PRIVATE cw_process Threads::Threads)
  add_test(NAME cw-process COMMAND cw-process-tests "$<TARGET_FILE:cw-process-probe>")
endif()
